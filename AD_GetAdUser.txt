# Filtert typische System- und Dienstkonten heraus
(Get-ADUser -Filter "Enabled -eq `$true" -Properties Mail | Where-Object {
    $_.Mail -ne $null -and $_.SamAccountName -notlike "SM_*" -and $_.SamAccountName -notlike "healthmail*"
}).Count

# Get user information: FirstName, LastName, SAMAccountName, UPN, DisplayName, EmailAddress, and Title
$users = Get-ADUser -Filter { Enabled -eq $true } -Property GivenName, Surname, SAMAccountName, UserPrincipalName, DisplayName, EmailAddress, Title, Department, Enabled, DistinguishedName  |
Select-Object DisplayName, GivenName, Surname, SAMAccountName, UserPrincipalName, EmailAddress, Title, Department, Enabled, DistinguishedName, @{Name="OU";Expression={$_.DistinguishedName -replace '^CN=.*?,OU=(.*?),DC=.*$','$1'}}
$users | Export-Csv "C:\install\ADExport\AD_User_Export.csv" -Encoding Unicode -NoTypeInformation -Delimiter ";"
